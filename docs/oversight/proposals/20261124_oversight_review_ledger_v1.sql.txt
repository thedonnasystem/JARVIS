-- Coordination: the 20261123_ prefix is reserved by the WS3 org-architecture branch; names here (oversight_*) were chosen not to
-- collide by name only. That branch's schema was NOT read.
--
-- Purpose: a database mirror of ops/oversight/ledger/reviews.jsonl, so the file has a second anchor that repository write access
-- cannot rewrite. The file stays the source; this table is a copy with its own append-only guard.

create table if not exists public.oversight_reviews (
  seq          bigint primary key,                       -- same seq as the file
  entry_id     text not null unique,
  entry_type   text not null check (entry_type in ('review', 'mechanical_check', 'import', 'owner_decision')),
  subject_sha  text not null check (subject_sha ~ '^[0-9a-f]{40}$'),
  subject_pr   integer,
  reviewer_id  text,
  author_id    text,
  recommendation text check (recommendation in ('APPROVE', 'REJECT', 'NEEDS_MORE_WORK', 'RECOMMEND_APPROVE')),
  prev_hash    text not null check (prev_hash ~ '^[0-9a-f]{64}$'),
  entry_hash   text not null unique check (entry_hash ~ '^[0-9a-f]{64}$'),
  entry        jsonb not null,                           -- the full ledger line. Must contain no client PII (reviews are about code).
  recorded_at  timestamptz not null default now(),
  check (entry_type = 'owner_decision' or reviewer_id is distinct from author_id)
);
alter table public.oversight_reviews enable row level security;         -- no policy: default deny
revoke all on public.oversight_reviews from public, anon, authenticated, service_role;

-- append-only: no update, delete or truncate by anyone, and each row must chain to the previous one
create or replace function public.oversight_reviews_guard() returns trigger language plpgsql set search_path = pg_catalog as $$
declare v_prev text; v_max bigint;
begin
  if tg_op <> 'INSERT' then raise exception 'oversight_reviews is append-only'; end if;
  select max(seq) into v_max from public.oversight_reviews;
  if new.seq is distinct from coalesce(v_max, 0) + 1 then raise exception 'oversight_reviews: seq % is not the next entry', new.seq; end if;
  select entry_hash into v_prev from public.oversight_reviews where seq = new.seq - 1;
  if new.prev_hash is distinct from coalesce(v_prev, repeat('0', 64)) then raise exception 'oversight_reviews: chain broken at seq %', new.seq; end if;
  return new;
end $$;
revoke all on function public.oversight_reviews_guard() from public, anon, authenticated;
create trigger oversight_reviews_guard before insert or update or delete on public.oversight_reviews for each row execute function public.oversight_reviews_guard();
create trigger oversight_reviews_no_truncate before truncate on public.oversight_reviews for each statement execute function public.oversight_reviews_guard();

-- Writer: the bot (service_role) may mirror review / mechanical / import entries. It can never mirror an owner decision.
create or replace function public.oversight_record_review(p_entry jsonb) returns bigint language plpgsql security definer set search_path = pg_catalog as $$
begin
  if p_entry->>'type' not in ('review', 'mechanical_check', 'import') then raise exception 'only the owner path records owner decisions'; end if;
  if p_entry ? 'owner_decision' or p_entry ? 'owner_approved' or p_entry ? 'decision' or p_entry ? 'signature' then raise exception 'owner fields are not allowed here'; end if;
  insert into public.oversight_reviews (seq, entry_id, entry_type, subject_sha, subject_pr, reviewer_id, author_id, recommendation, prev_hash, entry_hash, entry)
  values ((p_entry->>'seq')::bigint, p_entry->>'id', p_entry->>'type', p_entry->'subject'->>'sha', (p_entry->'subject'->>'pr')::integer,
          p_entry->'reviewer'->>'id', p_entry->'author'->>'id', coalesce(p_entry->>'recommendation', p_entry->>'verdict'),
          p_entry->>'prev_hash', p_entry->>'entry_hash', p_entry);
  return (p_entry->>'seq')::bigint;
end $$;
revoke all on function public.oversight_record_review(jsonb) from public, anon, authenticated;

-- Owner path: only the signed-in owner (existing pattern: public.ffm_require_owner(), which refuses service_role) can mirror a decision.
-- The Ed25519 signature is still verified off-database by ledger.py; this function only guarantees WHO inserted the row.
create or replace function public.oversight_record_owner_decision(p_entry jsonb) returns bigint language plpgsql security definer set search_path = pg_catalog as $$
begin
  perform public.ffm_require_owner();
  if p_entry->>'type' <> 'owner_decision' then raise exception 'not an owner decision'; end if;
  if not exists (select 1 from public.oversight_reviews where entry_id = p_entry->>'review_id') then raise exception 'unknown review'; end if;
  insert into public.oversight_reviews (seq, entry_id, entry_type, subject_sha, subject_pr, recommendation, prev_hash, entry_hash, entry)
  values ((p_entry->>'seq')::bigint, p_entry->>'id', 'owner_decision', p_entry->'subject'->>'sha', (p_entry->'subject'->>'pr')::integer, null,
          p_entry->>'prev_hash', p_entry->>'entry_hash', p_entry);
  return (p_entry->>'seq')::bigint;
end $$;
revoke all on function public.oversight_record_owner_decision(jsonb) from public, anon, authenticated, service_role;
